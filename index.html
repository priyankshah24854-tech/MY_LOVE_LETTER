const noBtn = document.getElementById('noBtn'); function moveButton(e) { if (e) { e.preventDefault(); e.stopPropagation(); } // Screen margins calculate karna const padding = 20; const maxX = window.innerWidth - noBtn.offsetWidth - padding; const maxY = window.innerHeight - noBtn.offsetHeight - padding; const randomX = Math.max(padding, Math.floor(Math.random() * maxX)); const randomY = Math.max(padding, Math.floor(Math.random() * maxY)); noBtn.style.position = 'fixed'; noBtn.style.left = randomX + 'px'; noBtn.style.top = randomY + 'px'; noBtn.style.zIndex = '9999'; } // Mobile touch events noBtn.addEventListener('touchstart', moveButton, { passive: false }); noBtn.addEventListener('pointerdown', moveButton); noBtn.addEventListener('mouseover', moveButton);
